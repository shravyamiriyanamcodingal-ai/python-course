
import random
import math

num1 = random.randint(1, 20)
num2 = random.randint(1, 20)

print("First random number:", num1)
print("Second random number:", num2)

print("Addition:", num1 + num2)
print("Subtraction:", num1 - num2)
print("Multiplication:", num1 * num2)

if num2 != 0:
    print("Division:", num1 / num2)

print("Power:", math.pow(num1, num2))
print("Square root of first number:", math.sqrt(num1))
print("Factorial of first number:", math.factorial(num1))
print("GCD:", math.gcd(num1, num2))
