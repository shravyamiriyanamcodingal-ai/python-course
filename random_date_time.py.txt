
import random
import datetime

start_date = datetime.datetime(2020, 1, 1)
end_date = datetime.datetime(2030, 12, 31, 23, 59, 59)

time_difference = end_date - start_date

random_seconds = random.randint(0, int(time_difference.total_seconds()))

random_datetime = start_date + datetime.timedelta(seconds=random_seconds)

print("Random date:", random_datetime.strftime("%d-%m-%Y"))
print("Random time:", random_datetime.strftime("%H:%M:%S"))
print("Random date and time:", random_datetime.strftime("%d-%m-%Y %H:%M:%S"))
